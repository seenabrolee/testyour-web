function applyResultToUI(nenKey, secondKey, thirdKey) {
            const n = nenTypes[nenKey];
            document.getElementById('result-title').innerText = n.name;
            document.getElementById('result-image').src = n.image;
            
            let desc = n.description;
            
            if (secondKey && thirdKey) {
                const s2 = nenTypes[secondKey];
                const s3 = nenTypes[thirdKey];
                desc += `<br><br><hr style="border:0; border-top:1px solid rgba(255,255,255,0.2); margin: 20px 0;">`;
                desc += `<strong>📊 내 안에 숨겨진 또 다른 계통</strong><br>`;
                desc += `당신의 응답을 분석한 결과, 아래 두 가지 계통의 성향도 강하게 나타납니다!<br><br>`;
                desc += `<strong>2순위: ${s2.name}</strong><br>`;
                desc += `• 나와의 궁합: <span style="color:#8b5cf6">${getSynergy(nenKey, secondKey)}</span><br><br>`;
                desc += `<strong>3순위: ${s3.name}</strong><br>`;
                desc += `• 나와의 궁합: <span style="color:#8b5cf6">${getSynergy(nenKey, thirdKey)}</span>`;
            }
            
            document.getElementById('result-desc').innerHTML = desc;
        }
